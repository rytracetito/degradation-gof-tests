
# Puissance et différents plans d'échantillonnage

```{r}
# fonction pour la construction de tobs

make_tobs <- function(n_obs, tmax = 50, p = c(4,3,2,1)) {
  k <- round(n_obs * p / sum(p))
  k[1] <- k[1] + (n_obs - sum(k))
  t1 <- seq(0, tmax/4, length.out = k[1])
  t2 <- seq(tmax/4, tmax/2, length.out = k[2] + 1)[-1]
  t3 <- seq(tmax/2, 3*tmax/4, length.out = k[3] + 1)[-1]
  t4 <- seq(3*tmax/4, tmax, length.out = k[4] + 1)[-1]
  t_obs <- c(t1, t2, t3, t4)
  t_obs <- unique(round(t_obs, 10))
  t_obs <- t_obs[t_obs > 0]
  return(t_obs)
}
```

## Fréquence décroissante

```{r}
# Plus de points au début 
set.seed(46)
nobs = 20
t1 <-c(0, make_tobs(nobs, tmax = tmax, p = c(4, 2, 1, 1)))

stripchart(t1,
           method="stack",
           pch=19,
           col="steelblue",
           main="fréquence décroissante",
           xlab="Temps",
           ylab="Observations")
```
